LangChain Integration

Drop-in memory adapter and runnable memory hooks for LangChain chains and agents.

1. Installation & Setup

bash
pip install langchain langchain-openai piyapi-memory
export PIYAPI_API_KEY="piy_live_sec_..."

2. Drop-in Memory Adapter

integration.py 
from langchain_openai import ChatOpenAI
from langchain.chains import ConversationChain
from piyapi.adapters.langchain import PiyChatMessageHistory

# Bind LangChain conversation history directly to PiyAPI cognitive substrate
history = PiyChatMessageHistory(
    user_id="user_enterprise_101",
    session_id="sess_langchain_01",
    api_key="your_api_key",
    surprise_threshold=0.05
)

llm = ChatOpenAI(model="gpt-4o")
chain = ConversationChain(llm=llm, memory=history)

response = chain.invoke({"input": "We have finalized our database migration to PostgreSQL 16."})
print(response["response"])
integration.ts 
import { ChatOpenAI } from "@langchain/openai";
import { ConversationChain } from "langchain/chains";
import { PiyChatMessageHistory } from "@piyapi/sdk/langchain";

const memory = new PiyChatMessageHistory({
  userId: "user_enterprise_101",
  sessionId: "sess_langchain_01",
  apiKey: process.env.PIYAPI_API_KEY!
});

const model = new ChatOpenAI({ modelName: "gpt-4o" });
const chain = new ConversationChain({ llm: model, memory });
const res = await chain.call({ input: "Finalized DB migration to PostgreSQL 16." });
console.log(res.response);

LangGraph Multi-Agent Persistence

Persistent graph checkpointer and cross-node episodic memory for LangGraph state machines.

1. Installation & Setup

2. Persistent StateGraph Checkpointer

integration.py 
from langgraph.graph import StateGraph, START, END
from piyapi.adapters.langgraph import PiyCheckpointer

# Initialize persistent memory checkpointer
checkpointer = PiyCheckpointer(
    user_id="analyst_team",
    workspace_id="ws_financial_reports"
)

builder = StateGraph(dict)
# Define research and drafting nodes
builder.add_node("research", lambda state: {"facts": "Ingested 12 earnings reports"})
builder.add_node("draft", lambda state: {"summary": "Q3 EBITDA up 18% YoY"})
builder.add_edge(START, "research")
builder.add_edge("research", "draft")
builder.add_edge("draft", END)

# Compile graph with persistent PiyAPI memory checkpointer
graph = builder.compile(checkpointer=checkpointer)
result = graph.invoke({"task": "Analyze Q3 tech earnings"}, config={"configurable": {"thread_id": "thread_44"}})
print("LangGraph Execution Output:", result)

});

from piyapi import PiyMemoryClient

app = FastAPI()

@app.post("/api/chat")
async def chat_handler(payload: dict):
    user_id = payload.get("user_id")
    messages = payload.get("messages")
    # Ingest turn into memory
    res = piy.memories.add(user_id=user_id, messages=messages)

integration.ts 
import { streamText } from 'ai';
import { openai } from '@ai-sdk/openai';
import { PiyMemoryClient } from '@piyapi/sdk';

const piy = new PiyMemoryClient({ apiKey: process.env.PIYAPI_API_KEY! });

export async function POST(req: Request) {
  const { messages, userId } = await req.json();

  // 1. Recall hot memories for user
  const context = await piy.memories.search({
    userId,
    query: messages[messages.length - 1].content,
    limit: 5
  });

  const systemPrompt = `You are a helpful assistant. Context:\n${context.results.map(r => r.content).join('\n')}`;

  // 2. Stream response
  const result = streamText({
    model: openai('gpt-4o'),
    system: systemPrompt,
    messages,
    async onFinish({ text }) {
      // 3. Persist completed turn asynchronously
      await piy.memories.add({
        userId,
        messages: [
          messages[messages.length - 1],
          { role: 'assistant', content: text }
        ]
      });
    }
  });

  return result.toDataStreamResponse();
}

2. Pipecat Memory Frame Processor

integration.py 
from pipecat.processors.frame_processor import FrameDirection, FrameProcessor
from piyapi import PiyMemoryClient

class PiyMemoryProcessor(FrameProcessor):
    def __init__(self, user_id: str):
        super().__init__()
        self.user_id = user_id
        self.piy = PiyMemoryClient()

    async def process_frame(self, frame, direction: FrameDirection):
        await super().process_frame(frame, direction)
        # Intercept completed transcribed user text
        if hasattr(frame, "text") and frame.text:
            self.piy.memories.add(
                user_id=self.user_id,
                messages=[{"role": "user", "content": frame.text}]
            )
        await self.push_frame(frame, direction)

CDC Webhooks & Event Ingestion

1. HMAC Signature Verification

python
import hmac, hashlib

def verify_piyapi_webhook(raw_payload: bytes, signature_header: str, secret: str) -> bool:
    computed = hmac.new(secret.encode(), raw_payload, hashlib.sha256).hexdigest()
    return hmac.compare_digest(f"sha256={computed}", signature_header)

CREATE TABLE piy_memories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    metadata JSONB
);

CREATE INDEX idx_piy_valid_time ON piy_memories USING gist (valid_time);
